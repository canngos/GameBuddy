# The service tier is range-pinned; the pickle tier is pinned exactly.
#
# The artefact is a pickle of fitted scikit-learn estimators, so the version that writes
# it and the version that reads it have to agree. Unpickling across versions is undefined
# behaviour, and the way it usually surfaces is not an exception — it is a model that
# loads, answers every request, and ranks wrongly. A range for the three libraries whose
# objects are inside the pickle is not a pin at all: `pip install` a month apart produces
# two different resolutions and nothing warns you. So they are `==`, and
# `gamebuddy_model.version` stamps what was used into the artefact so the API can check
# at load time rather than leaving it to be noticed.
#
# Raising these is a deliberate act: bump the version, retrain, redeploy the artefact with
# the image. Do not bump one without the other.
fastapi>=0.115,<1.0
uvicorn[standard]>=0.32,<1.0
pydantic>=2.9,<3.0
pandas>=2.2,<4.0

numpy==2.4.1
scikit-learn==1.9.0
scipy==1.18.0

# nbformat and nbconvert are gone: they existed only so an HTTP endpoint could execute a
# Jupyter notebook in-process. Training is a batch job now.
#
# psycopg is deliberately NOT here. `python -m gamebuddy_model export` needs it, but the
# serving container never talks to Postgres — it loads a pickle — and shipping a database
# driver plus credentials handling into an image that has no use for either is avoidable
# surface. It lives in requirements-tools.txt, and export.py imports it inside the
# function so the package stays importable without it.

# --- image moderation ------------------------------------------------------
# Nothing an account uploads is shown to another account until it has been screened, and
# a paid vision API would bill per image forever — so the classifier is self-hosted here.
#
# torch is the CPU build. The default PyPI wheel bundles CUDA and is several gigabytes for
# a service that will never see a GPU; the Dockerfile installs these from PyTorch's CPU
# index for that reason. If you install requirements by hand, do the same:
#
#   pip install -r requirements.txt --extra-index-url https://download.pytorch.org/whl/cpu
#
# python-multipart is not optional despite looking it: FastAPI's UploadFile silently
# refuses to parse a multipart body without it, and the failure is a 422 on a request that
# is perfectly well formed.
torch>=2.4,<3.0
transformers>=4.44,<6.0
pillow>=10.4,<12.0
python-multipart>=0.0.9,<1.0
